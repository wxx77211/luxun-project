    function addMarkerToMap(map,marker,mapType){
        const markerDiv = document.createElement('div'); markerDiv.className='map-marker'; markerDiv.style.left=marker.left; markerDiv.style.top=marker.top;

        const locChars = locationMapping[mapType][marker.name] || [];
